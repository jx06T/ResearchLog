```python
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

        self.bottleneck_size = 4      # n size (1, 2, 4, 8, 16...)

        self.lambda_l1 =  1     # L1 Loss 權重 (L1 與 D-loss 的比例)

        self.aug_scale = (0.8, 1.3)   # 增強縮放範圍
        self.aug_translate = (0.3, 0.3) # 增強平移範圍
        self.aug_mask_prob = 0    # 遮罩機率
        self.aug_degrees = 0

        # GAN 基礎參數
        self.nz = 256            # 噪聲維度 感覺影響不大
        self.ngf = 64            # 生成器基礎特徵圖數
        self.ndf = 64            # 判別器基礎特徵圖數
        self.image_size = 64     # 圖像大小
        self.nc = 1              # 通道數(黑白)
        
        self.lambda_gp = 10      # WGAN-GP 梯度懲罰 依照原論文推薦
        self.batch_size = 32
        self.betas = (0.0, 0.9)  # Adam optimizer 參數依照原WGAN-GP論文推薦
        
        self.lr = 0.0004
        self.num_epochs = 200    
        self.n_critic = 1        # 判別器訓練次數 / 生成器訓練次數
```
